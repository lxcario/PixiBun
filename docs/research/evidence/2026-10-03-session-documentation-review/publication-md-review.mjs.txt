import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
import { ownedMarkdown } from '../../scripts/docContracts.mjs';
const root=process.cwd();
const previous=JSON.parse(fs.readFileSync('docs/research/evidence/2026-10-02-feature-repairs-phase5/markdown-review.json','utf8'));
const files=ownedMarkdown(root);
const current=/^(?:AGENTS|ARCHITECTURE|BENCHMARKS|CONTRIBUTING|DESIGN|MAIN|PIXIBUN_DEEP_DIVE|README|SECURITY|SKILL)\.md$|^docs\/(?:README|features|current-status)\.md$|^docs\/(?:contracts|guides)\//;
const records=files.map(file=>{
 const content=fs.readFileSync(file,'utf8');
 const old=previous.files.find(row=>row.path===file);
 const scope=file.startsWith('.agents/')?'vendor instruction':current.test(file)?'operating reference':file.startsWith('docs/plans/')||file==='V2_VISION.md'?'plan or proposal':'dated record or historical ledger';
 const risks=content.split(/\r?\n/).flatMap((line,index)=> /16\.3\.4|pixibun\.art|phase [567]|underway|in progress|still running|pending|ArrayBuffer|10,001|10_001|archive|rollback|Godot|FREE|practice|working tree|full suite|full-suite|reviewed|live inventory|current inventory/i.test(line)?[{line:index+1,text:line}]:[]);
 return {path:file,scope,bytes:Buffer.byteLength(content),sha256:crypto.createHash('sha256').update(content).digest('hex'),priorScope:old?.scope??null,risks};
});
fs.mkdirSync('artifacts/feature-repairs/publication',{recursive:true});
fs.writeFileSync('artifacts/feature-repairs/publication/markdown-intake.json',JSON.stringify({recordedAt:new Date().toISOString(),files:records},null,2)+'\n');
