import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
import {spawnSync} from 'node:child_process';
import {ownedMarkdown} from '../../scripts/docContracts.mjs';
const root=process.cwd();
const dest='docs/research/evidence/2026-10-03-session-documentation-review';
fs.mkdirSync(dest,{recursive:true});
const hash=file=>crypto.createHash('sha256').update(fs.readFileSync(file)).digest('hex');
const intake=JSON.parse(fs.readFileSync('artifacts/feature-repairs/publication/markdown-intake.json','utf8'));
const corrections={
 'ARCHITECTURE.md':'Added count-before-payload admission, physical/logical MIME boundary and conditional archive availability.',
 'docs/contracts/storage.md':'Removed duplicate archive-admission section, preserving its first anchor and limits.',
 'docs/guides/artwork-safety.md':'Removed repeated overflow paragraph; retained current in-dialog warning and phase6 pictures.',
 'docs/contracts/security-boundaries.md':'Included latest archive-admission and logical-field safeguards with source/evidence links.',
 'docs/guides/deployment.md':'Documented actual conditional main CI/deploy trigger and unverified independent host integration.',
 'docs/guides/release-readiness.md':'Added publication-trigger review before candidate push.',
 'docs/plans/README.md':'Marked seven repair phases complete locally; production approval stays pending.',
 'docs/research/issue-registry.md':'Updated session checkpoint/current authority; kept research and release gates open.',
 'docs/features.md':'Reconciled session review date; screenshot capture dates remain2October.',
 'docs/README.md':'Linked final publication review alongside earlier dated audit/repair records.',
 'docs/current-status.md':'Linked final docs-only reconciliation without changing executed application scope.',
 'docs/research/markdown-maintenance-map.md':'Kept original52-file intake map and linked final expanded publication review.',
 'README.md':'Retained supplied logo, warm beginner wording and current Cozy captures; current features/limits link to canonical guides.',
 'SECURITY.md':'Retained user-supplied private reporting address, pinned patch evidence and dated audit limitations.',
 'docs/research/2026-10-03-session-documentation-review.md':'New final publication review; source publication, deployed identity and production approval distinguished.'
};
const records=ownedMarkdown(root).map(file=>{
 const old=intake.files.find(row=>row.path===file);
 const text=fs.readFileSync(file,'utf8');
 const scope=old?.scope??'dated publication record';
 const disposition=corrections[file]??(scope==='vendor instruction'?'Preserved upstream instruction; local-only testing preference remains recorded.':scope==='operating reference'?'Reviewed session-affected claims and source/evidence destinations; retained current bounded behavior and qualifications.':scope==='plan or proposal'?'Preserved proposal/historical phase scope; implementation and remaining qualification stay distinct.':'Preserved dated revision, counts, failures, captures and stated scope; later evidence is linked from canonical current status.');
 return {path:file,scope,bytes:Buffer.byteLength(text),sha256:hash(file),changedDuringFinalReview:old?old.sha256!==hash(file):true,disposition};
});
fs.writeFileSync(`${dest}/markdown-review.json`,JSON.stringify({reviewedAt:new Date().toISOString(),localDate:'2026-10-03',timeZone:'Europe/Istanbul',ownedMarkdownFiles:records.length,scope:'All publishable Markdown classified and session-affected claims reconciled; not a line-by-line independent certification of all algorithms or workflows.',exclusions:['ignored private transcripts/proposals','node_modules and generated runtimes','separate checkouts'],files:records},null,2)+'\n');
const source=JSON.parse(fs.readFileSync('docs/research/evidence/2026-10-02-feature-repairs-phase7/source-final.json','utf8'));
for(const file of source.files)if(hash(file.path)!==file.sha256)throw Error(`Changed tested source:${file.path}`);
const seals=[];
for(const name of fs.readdirSync('docs/research/evidence')){
 const dir=`docs/research/evidence/${name}`;
 const manifest=`${dir}/manifest.json`;
 if(!fs.existsSync(manifest))continue;
 const list=JSON.parse(fs.readFileSync(manifest,'utf8'));
 if(!Array.isArray(list))continue;
 for(const file of list)if(hash(`${dir}/${file.path}`)!==file.sha256)throw Error(`Changed sealed evidence:${dir}/${file.path}`);
 seals.push({directory:dir,matchingHashes:list.length});
}
const pics=[];
for(const category of ['docs/assets/screenshots','docs/assets/readme']){
 for(const name of fs.readdirSync(category)){
  const file=`${category}/${name}/run.json`;
  if(!fs.existsSync(file))continue;
  const metadata=JSON.parse(fs.readFileSync(file,'utf8'));
  pics.push({path:file,sha256:hash(file),note:'Capture metadata retained; final Markdown checker verifies image destinations.'});
 }
}
const files=spawnSync('git',['ls-files','--cached','--others','--exclude-standard','-z'],{encoding:'utf8'}).stdout.split('\0').filter(Boolean);
const oversized=files.filter(file=>fs.statSync(file).size>=100*1024*1024);
if(oversized.length)throw Error(`Oversized publication files:${oversized}`);
const env={...process.env,PATH:`${path.dirname(process.execPath)}${path.delimiter}${process.env.PATH}`};
const commands=[];
const checksFile=`${dest}/checks.json`;
if(!fs.existsSync(checksFile))fs.writeFileSync(checksFile,JSON.stringify({status:'Checks executing; this placeholder is not a verdict.'},null,2)+'\n');
function run(name,command,args){
 const beganAt=new Date().toISOString();
 const result=spawnSync(command,args,{env,encoding:'utf8',maxBuffer:16*1024*1024});
 if(fs.existsSync(`${dest}/${name}.log`)&&!fs.existsSync(`${dest}/${name}-initial.log`))fs.copyFileSync(`${dest}/${name}.log`,`${dest}/${name}-initial.log`);
 fs.writeFileSync(`${dest}/${name}.log`,`${result.stdout??''}${result.stderr??''}`);
 commands.push({command:[command,...args],exitCode:result.status,beganAt,completedAt:new Date().toISOString(),log:`${name}.log`});
 if(result.error||result.status!==0)throw result.error??Error(`${name} failed`);
 console.log(`${name} passed.`);
}
run('docs-final',process.execPath,[path.join(path.dirname(process.execPath),'node_modules/npm/bin/npm-cli.js'),'run','verify:docs']);
run('diff-check','git',['diff','--check']);
fs.writeFileSync(`${dest}/checks.json`,JSON.stringify({checkedAt:new Date().toISOString(),commands,matchingTestedSourceFiles:source.files.length,sealedEvidence:seals,captureCheckpoints:pics,oversizedPublicationFiles:oversized,scope:'Documentation checks, exact unchanged phase7 source hashes, all retained evidence manifests; no new browser/security/device/release verdict.'},null,2)+'\n');
console.log(`Reviewed ${records.length} Markdown files; ${source.files.length} tested source hashes and ${seals.length} sealed evidence sets intact.`);
