const fs = require('node:fs');
const path = require('node:path');
const cp = require('node:child_process');
const crypto = require('node:crypto');
const sha = b => crypto.createHash('sha256').update(b).digest('hex');
const records = [];
for (const name of ['2026-10-02', '2026-10-02-remediation', ...[2, 3, 4, 5, 6].map(n => `2026-10-02-feature-repairs-phase${n}`)]) {
  const folder = `docs/research/evidence/${name}`;
  const manifest = JSON.parse(fs.readFileSync(`${folder}/manifest.json`));
  for (const entry of manifest) if (sha(fs.readFileSync(path.join(folder, entry.path))) !== entry.sha256) throw new Error(`Changed prior evidence: ${folder}/${entry.path}`);
  records.push({ directory: folder, unchangedFiles: manifest.length });
}
const tracked = cp.execFileSync('git', ['ls-files', '-z'], { encoding: 'utf8' }).split('\0').filter(p => p.endsWith('.png'));
for (const file of tracked) if (sha(cp.execFileSync('git', ['show', `HEAD:${file}`], { maxBuffer: 64 * 1024 * 1024 })) !== sha(fs.readFileSync(file))) throw new Error(`Changed tracked image: ${file}`);
for (const folder of ['docs/assets/readme/2026-10-02', 'docs/assets/screenshots/2026-10-02-docs', ...['2026-10-02-feature-guide', ...[1, 2, 3, 4, 5, 6].map(n => `2026-10-02-feature-repairs-phase${n}`), '2026-10-02-feature-repairs-phase5-welcome'].map(n => `docs/assets/screenshots/${n}`)]) {
  const metadata = JSON.parse(fs.readFileSync(`${folder}/run.json`));
  const captures = metadata.captures ?? metadata.images;
  for (const capture of captures) if (sha(fs.readFileSync(path.join(folder, capture.path ?? capture.file))) !== capture.sha256) throw new Error(`Changed prior capture: ${folder}`);
  records.push({ directory: folder, matchingCaptureHashes: captures.length });
}
const snapshot = JSON.parse(fs.readFileSync('artifacts/feature-repairs/phase7/source-final.json'));
for (const file of snapshot.files) if (sha(fs.readFileSync(file.path)) !== file.sha256) throw new Error(`Changed frozen source: ${file.path}`);
const additional = JSON.parse(fs.readFileSync('artifacts/feature-repairs/phase7/additional-build-inputs.json'));
for (const file of additional.files) if (sha(fs.readFileSync(file.path)) !== file.sha256) throw new Error(`Changed additional build input: ${file.path}`);
const build = JSON.parse(fs.readFileSync('artifacts/feature-repairs/phase7/environment-build.json'));
for (const file of build.productionBuildFiles) if (sha(fs.readFileSync(file.path)) !== file.sha256) throw new Error(`Changed production build file: ${file.path}`);
const copy = JSON.parse(fs.readFileSync('artifacts/feature-repairs/phase7/editor-app-source.json'));
const copyChanges = [];
if (fs.existsSync(copy.path)) {
  for (const file of copy.copiedFiles) {
    const actual = sha(fs.readFileSync(path.join(copy.path, file.path)));
    if (actual !== file.sha256) copyChanges.push({ path: file.path, before: file.sha256, after: actual });
  }
}
const result = { kind: 'hash-integrity-not-workflow-verdict', sourceFilesUnchanged: snapshot.files.length, additionalInputsUnchanged: additional.files.length, productionBuildFilesUnchanged: build.productionBuildFiles.length, trackedPngsUnchanged: tracked.length, isolatedCopyChanges: copyChanges, records };
fs.writeFileSync('artifacts/feature-repairs/phase7/integrity.json', JSON.stringify(result, null, 2) + '\n');
console.log(`Verified ${snapshot.files.length} frozen files, ${tracked.length} tracked PNGs, seven prior sealed sets and ten capture checkpoints.`);
