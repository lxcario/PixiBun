const fs = require('node:fs');
const cp = require('node:child_process');
const crypto = require('node:crypto');
const path = require('node:path');
const sha = b => crypto.createHash('sha256').update(b).digest('hex');
const folder = 'artifacts/feature-repairs/phase7';
fs.mkdirSync(folder, { recursive: true });
const mode = process.argv[2];
const label = process.argv[3] ?? '';
const snapshotName = label ? 'source-final' : 'source-before';
if (mode === 'freeze') {
  const paths = cp.execFileSync('git', ['ls-files', '--cached', '--others', '--exclude-standard', '-z', '--', 'src', 'tests', 'scripts', 'package.json', 'package-lock.json', 'next.config.ts', 'playwright.config.ts', 'playwright.production.config.ts', 'AGENTS.md'], { encoding: 'utf8' }).split('\0').filter(Boolean);
  const files = [...new Set(paths)].sort().filter(p => fs.statSync(p).isFile()).map(path => ({ path, sha256: sha(fs.readFileSync(path)) }));
  const record = { recordedAt: new Date().toISOString(), baseRevision: cp.execFileSync('git', ['rev-parse', 'HEAD'], { encoding: 'utf8' }).trim(), files };
  const previous = JSON.parse(fs.readFileSync('docs/research/evidence/2026-10-02-feature-repairs-phase6/source-final.json'));
  record.changesFromPhase6 = files.filter(f => previous.files.find(p => p.path === f.path)?.sha256 !== f.sha256);
  record.cacheVersion = `phase7-local-${sha(JSON.stringify(files)).slice(0, 16)}`;
  fs.writeFileSync(`${folder}/${snapshotName}.json`, JSON.stringify(record, null, 2) + '\n');
  fs.writeFileSync(`${folder}/git-status-${label || 'before'}.txt`, cp.execFileSync('git', ['status', '--short'], { encoding: 'utf8' }));
  console.log(`Frozen ${files.length} files; ${record.changesFromPhase6.length} changes from phase 6. Cache version ${record.cacheVersion}`);
  process.exit(0);
}
const snapshot = JSON.parse(fs.readFileSync(`${folder}/${snapshotName}.json`));
for (const f of snapshot.files) if (sha(fs.readFileSync(f.path)) !== f.sha256) throw new Error(`Changed frozen source: ${f.path}`);
const env = { ...process.env, PATH: `${path.dirname(process.execPath)}${path.delimiter}${process.env.PATH}`, SW_CACHE_VERSION: snapshot.cacheVersion };
delete env.CI; delete env.VITEST_MAX_WORKERS;
let args;
if (mode === 'gate') args = ['scripts/runQualityGate.mjs', '--with-build'];
else if (mode === 'editor') { env.PLAYWRIGHT_HTML_OUTPUT_DIR = `${folder}/editor${label}-report`; args = ['node_modules/@playwright/test/cli.js', 'test', ...(label ? ['--config', `${folder}/playwright.editor.config.cjs`] : []), '--workers=1', '--retries=0', '--trace=retain-on-failure', `--output=${folder}/editor${label}`]; }
else if (mode === 'production') { env.PIXIBUN_PRODUCTION_PORT = '3118'; args = ['node_modules/@playwright/test/cli.js', 'test', '--config', 'playwright.production.config.ts']; }
else throw new Error('Unknown mode');
const log = fs.openSync(`${folder}/${mode}${label}.log`, 'wx');
fs.writeSync(log, `Started ${new Date().toISOString()}\nNode ${process.version}; cache ${env.SW_CACHE_VERSION}; CI absent; VITEST_MAX_WORKERS absent\nCommand: node ${args.join(' ')}\n`);
const start = Date.now();
const child = cp.spawn(process.execPath, args, { env, stdio: ['ignore', log, log] });
child.on('error', e => { fs.writeSync(log, String(e)); fs.closeSync(log); console.error(e); process.exitCode = 1; });
child.on('exit', (code, signal) => {
  const result = { mode, label, args, cacheVersion: env.SW_CACHE_VERSION, exitCode: code, signal, elapsedSeconds: (Date.now() - start) / 1000, completedAt: new Date().toISOString() };
  fs.writeSync(log, `\n${JSON.stringify(result)}\n`); fs.closeSync(log);
  fs.writeFileSync(`${folder}/${mode}${label}-result.json`, JSON.stringify(result, null, 2) + '\n');
  console.log(JSON.stringify(result)); process.exitCode = code ?? 1;
});
