		/**
		* Defensive detector for IndexedDB QuotaExceededError across all major browsers.
		* Handles Chromium (name: 'QuotaExceededError'), WebKit/Safari (code: 22 / DOMException),
		* and Firefox storage exhaustion variants.
		*/
		static isQuotaExceeded(err) {
			if (!err || typeof err !== "object") return false;
			const error = err;
			if (error.name === "QuotaExceededError") return true;
			if (error.code === 22) return true;
			if (typeof error.message === "string" && error.message.toLowerCase().includes("quota")) return true;
			return false;
		}
		/**
		* Queries the browser StorageManager estimate defensively.
		* Returns safe fallbacks if the API is unsupported or rejects.
		*/
		static async queryStorageEstimate() {
			if (typeof navigator !== "undefined" && navigator.storage && typeof navigator.storage.estimate === "function") try {
				const estimate = await navigator.storage.estimate();
				const usageBytes = estimate.usage ?? 0;
				const quotaBytes = estimate.quota ?? 0;
				return {
					usageBytes,
					quotaBytes,
					usagePercent: quotaBytes > 0 ? usageBytes / quotaBytes * 100 : 0
				};
			} catch {}
			return {
				usageBytes: 0,
				quotaBytes: 0,
				usagePercent: 0
			};
		}

		* Emergency cache eviction of non-critical telemetry and diagnostics tables.

	/**
	* Computes the exact UTF-8 byte length of a string without allocating memory on the heap.
	* Handles single-byte ASCII, multi-byte code points, and UTF-16 surrogate pairs.
	*/
	function getUtf8ByteLength(str) {
		let bytes = 0;
		const len = str.length;
		for (let i = 0; i < len; i++) {
			const code = str.charCodeAt(i);
			if (code <= 127) bytes += 1;
			else if (code <= 2047) bytes += 2;
			else if (code >= 55296 && code <= 56319) {
				if (i + 1 < len) {
					const next = str.charCodeAt(i + 1);
					if (next >= 56320 && next <= 57343) {
						bytes += 4;
						i++;
						continue;
					}
				}
				bytes += 3;
			} else bytes += 3;
		}
		return bytes;
	}

		static sanitizeObject(obj, traversalBudget, currentDepth = 1, maxDepth = SafeJson.MAX_RECURSION_DEPTH) {
			if (traversalBudget !== void 0) {
				traversalBudget.count += 1;
				if (traversalBudget.count > traversalBudget.max) throw new SafeJsonTraversalLimitError(traversalBudget.count, traversalBudget.max);
			}
			if (currentDepth > maxDepth) throw new SafeJsonNestingDepthError(currentDepth, maxDepth, "with-detail");
			if (obj === null || typeof obj !== "object") return obj;
			if (Array.isArray(obj)) {
				if (traversalBudget !== void 0) {
					const remaining = traversalBudget.max - traversalBudget.count;
					if (obj.length > remaining) {
						traversalBudget.count = traversalBudget.max + 1;
						throw new SafeJsonTraversalLimitError(traversalBudget.max + 1, traversalBudget.max);
					}
					const arr = [];
					for (let i = 0; i < obj.length; i++) arr.push(this.sanitizeObject(obj[i], traversalBudget, currentDepth + 1, maxDepth));
					return arr;
				}
				const arr = new Array(obj.length);
				for (let i = 0; i < obj.length; i++) arr[i] = this.sanitizeObject(obj[i], void 0, currentDepth + 1, maxDepth);
				return arr;
			}
			const clean = {};
			for (const key of Object.keys(obj)) {
				if (key === "__proto__" || key === "constructor" || key === "prototype") continue;
				clean[key] = this.sanitizeObject(obj[key], traversalBudget, currentDepth + 1, maxDepth);
			}
			return clean;
		}

	/**
	* resourceBudget.ts
	*
	* Central aggregate project resource calculation and validation envelope for PixiBun.
	*
	* Initial conservative project-complexity ceiling. This is a validation bound based
	* on base/raw RGBA footprint, not a guarantee of total browser memory consumption.
	*
	* Ensures individually valid dimensions (e.g. 2048x2048 canvas, 500 frames, 100 layers)
	* cannot combine into an unmaterializable project (e.g. 50,000 layers = 781 GiB).
	*
	* Evaluates worst-case materializable raster footprint before allocation, import,
	* hydration, or P2P structure application.
	*/
	/**
	* Multiplies two non-negative safe integers, returning null if the result would exceed Number.MAX_SAFE_INTEGER.
	*/
	function safeMultiply(a, b) {
		if (!Number.isSafeInteger(a) || !Number.isSafeInteger(b) || a < 0 || b < 0) return null;
		if (a === 0 || b === 0) return 0;
		if (a > Math.floor(Number.MAX_SAFE_INTEGER / b)) return null;
		const result = a * b;
		return Number.isSafeInteger(result) ? result : null;
	}
	/**
	* Calculates aggregate worst-case materializable resource usage across all layers and frames.
	*/
	function calculateProjectResourceUsage(width, height, frames) {
		const frameCount = frames.length;
		let layerCount = 0;
		let rasterLayerCount = 0;
		for (const frame of frames) if (Array.isArray(frame.layers)) {
			layerCount += frame.layers.length;
			for (const layer of frame.layers) if (typeof layer === "object" && layer !== null) rasterLayerCount += 1;
		}
		const pxPerFrame = safeMultiply(width, height);
		const materializableRasterPixels = pxPerFrame !== null ? safeMultiply(layerCount, pxPerFrame) ?? Number.MAX_SAFE_INTEGER : Number.MAX_SAFE_INTEGER;
		const baseRawRgbaBytes = safeMultiply(materializableRasterPixels, 4) ?? Number.MAX_SAFE_INTEGER;
		return {
			frameCount,
			layerCount,
			rasterLayerCount,
			materializableRasterPixels,
			baseRawRgbaBytes
		};
	}
	/**
	* Validates aggregate materializable resources against the initial conservative
	* project-complexity ceiling. Rejects pathological allocations before any buffer creation.
	*/
	function validateProjectResourceBudget(width, height, frames) {
		const usage = calculateProjectResourceUsage(width, height, frames);
		if (usage.layerCount > WORKSPACE_ENVELOPE.MAX_PROJECT_TOTAL_LAYERS) return {
			ok: false,
			error: `Project exceeds total layer limit (${usage.layerCount.toLocaleString()} layers across all frames; maximum allowed is ${WORKSPACE_ENVELOPE.MAX_PROJECT_TOTAL_LAYERS.toLocaleString()}).`,
			usage
		};
		if (usage.materializableRasterPixels > WORKSPACE_ENVELOPE.MAX_PROJECT_AGGREGATE_PIXELS) {
			const mpx = (usage.materializableRasterPixels / 1e6).toFixed(1);
			const maxMpx = (WORKSPACE_ENVELOPE.MAX_PROJECT_AGGREGATE_PIXELS / 1e6).toFixed(1);
			return {
				ok: false,
				error: `Project exceeds conservative resource complexity ceiling (${mpx} MP worst-case footprint, ~${Math.round(usage.baseRawRgbaBytes / 1048576)} MiB base RGBA; maximum allowed is ${maxMpx} MP).`,
				usage
			};
		}
		return {
			ok: true,
			usage
		};
	}

				reasonCode: "INSPECTION_CEILING_EXCEEDED"
			};
			if (input.layers.length > WORKSPACE_ENVELOPE.MAX_CANVAS_LAYERS) frameExceedsEnvelope = true;
		}
		const layers = [];
		for (let i = 0; i < input.layers.length; i++) {
			const rawLayer = input.layers[i];
			if (!isPlainObject(rawLayer)) return {
				error: `Frame ${frameIndex + 1}, layer ${i + 1} is malformed.`,
				reasonCode: "INVALID_LAYER_STRUCTURE"
			};
			if (typeof rawLayer.id !== "string" || rawLayer.id.length === 0 || rawLayer.id.length > 128) return {
				error: `Frame ${frameIndex + 1}, layer ${i + 1} has an invalid identifier.`,
				reasonCode: "INVALID_LAYER_STRUCTURE"
			};
			if (typeof rawLayer.name !== "string" || rawLayer.name.length > 128) return {
				error: `Frame ${frameIndex + 1}, layer ${i + 1} has an invalid name.`,
				reasonCode: "INVALID_LAYER_STRUCTURE"
			};
			if (typeof rawLayer.visible !== "boolean" || typeof rawLayer.locked !== "boolean") return {
				error: `Frame ${frameIndex + 1}, layer ${i + 1} has corrupted visibility or lock flags.`,
				reasonCode: "INVALID_LAYER_STRUCTURE"
			};
			if (!isUnitInterval(rawLayer.opacity)) return {

				reasonCode: "INVALID_LAYER_STRUCTURE"
			};

			try {

					reasonCode: "INVALID_LAYER_STRUCTURE"
				};
			}

				error: `Frame ${frameIndex + 1}, layer ${i + 1} ${effect}`,
				reasonCode: "INVALID_LAYER_STRUCTURE"
			};

	/**
	* Upgrades legacy project payloads (such as v1 files) to the current format specification
	* by backfilling safe default values for newly introduced fields (clips, customRamps,
	* onionSkin settings, layer blendMode, opacity, lock flags, and effect recipes).
	*/
	function migrateProjectPayload(raw) {
		const migrated = { ...raw };
		if (!Array.isArray(migrated.customRamps)) migrated.customRamps = [];
		if (!Array.isArray(migrated.clips)) migrated.clips = [];
		if (typeof migrated.loopMode !== "string" || !LOOP_MODES.has(migrated.loopMode)) migrated.loopMode = "repeat";
		if (!isPlainObject(migrated.onionSkin)) migrated.onionSkin = {
			enabled: false,
			prevFrames: 1,
			nextFrames: 1,
			opacity: .35,
			prevColor: "#7fae6b",
			nextColor: "#c792df"
		};
		if (Array.isArray(migrated.frames)) migrated.frames = migrated.frames.map((f) => {
			if (!isPlainObject(f)) return f;
			const frameObj = { ...f };
			if (frameObj.durationMs === void 0) frameObj.durationMs = 100;
			if (Array.isArray(frameObj.layers)) frameObj.layers = frameObj.layers.map((l) => {
				if (!isPlainObject(l)) return l;
				const layerObj = { ...l };
				if (layerObj.visible === void 0) layerObj.visible = true;
				if (layerObj.locked === void 0) layerObj.locked = false;
				if (layerObj.opacity === void 0) layerObj.opacity = 1;
				if (layerObj.blendMode === void 0) layerObj.blendMode = "source-over";
				return layerObj;
			});
			return frameObj;
		});
		migrated.version = 3;
		return migrated;
	}

					actual: totalLayerCount,
					allowed: WORKSPACE_ENVELOPE.MAX_PROJECT_TOTAL_LAYERS
				});
				reasonCodes.push("ENVELOPE_MAX_TOTAL_LAYERS");
			}
			if (materializableRasterPixels > WORKSPACE_ENVELOPE.MAX_PROJECT_AGGREGATE_PIXELS) {
				const mpx = (materializableRasterPixels / 1e6).toFixed(1);
				const maxMpx = (WORKSPACE_ENVELOPE.MAX_PROJECT_AGGREGATE_PIXELS / 1e6).toFixed(1);

					message: `Project exceeds conservative resource complexity ceiling (${mpx} MP worst-case footprint, ~${mib} MiB base RGBA; maximum allowed is ${maxMpx} MP).`,
					actual: materializableRasterPixels,
					allowed: WORKSPACE_ENVELOPE.MAX_PROJECT_AGGREGATE_PIXELS
				});
				reasonCodes.push("ENVELOPE_MAX_MEGAPIXELS");
			}
		}
		return {
			ok: true,
			value: validatedProject,
			violations,
			reasonCodes
		};
	}

				released = true;
			}
		};

	/**
	* recoveryClassifier.ts
	*
	* Metadata-only Project Classifier for Restricted Recovery Mode.
	*
	* Inspects untrusted project JSON payloads without allocating canvas buffers,
	* decoding raster layers, or performing IndexedDB writes.
	*
	* Note on Allocation:
	* classifyProjectMetadata operates on project metadata without raster/image decoding,
	* but JSON.parse necessarily creates in-memory AST objects. SafeJson bounds this with
	* MAX_METADATA_NODES during its prototype-pollution sanitization pass.
	*/
	/**
	* Classifies an untrusted project metadata JSON payload into one of five states:
	* 1. 'valid'
	* 2. 'recoverable-envelope-exceeded'
	* 3. 'malformed'
	* 4. 'unsupported-version'
	* 5. 'unsafe-content'
	*
	* Never performs canvas/bitmap/Base64 decoding or IndexedDB mutations.
	*/
	function classifyProjectMetadata(rawJson) {
		if (typeof rawJson !== "string") return {
			status: "malformed",
			reasonCode: "INVALID_ROOT_TYPE",
			message: "Project payload must be a string."
		};
		let validation;
		try {
			validation = validateProjectJsonWithPolicy(rawJson, { envelopeMode: "collect-recoverable" }, { maxTraversedNodes: RECOVERY_INSPECTION_CEILINGS.MAX_METADATA_NODES });
		} catch (err) {
			if (err instanceof SafeJsonPayloadLimitError) return {
				status: "unsafe-content",
				reasonCode: "SAFE_JSON_PAYLOAD_LIMIT",
				message: err.message
			};
			if (err instanceof SafeJsonNestingDepthError) return {
				status: "unsafe-content",
				reasonCode: "SAFE_JSON_NESTING_DEPTH",
				message: err.message
			};
			if (err instanceof SafeJsonTraversalLimitError) return {
				status: "unsafe-content",
				reasonCode: "SAFE_JSON_TRAVERSAL_LIMIT",
				message: err.message
			};
			if (err instanceof SafeJsonSyntaxError || err instanceof SyntaxError) return {
				status: "malformed",
				reasonCode: "SAFE_JSON_SYNTAX_ERROR",
				message: "The project file contains invalid JSON syntax."
			};
			return {
				status: "malformed",
				reasonCode: "MALFORMED_JSON_SYNTAX",
				message: err instanceof Error ? err.message : "Failed to parse project JSON."
			};
		}
		if (!validation.ok) {
			if (validation.reasonCode === "UNSUPPORTED_FUTURE_VERSION") return {
				status: "unsupported-version",
				reasonCode: "UNSUPPORTED_FUTURE_VERSION",
				foundVersion: typeof validation.foundVersion === "number" ? validation.foundVersion : 4,
				supportedVersion: 3,
				message: validation.error
			};
			if (validation.reasonCode === "INSPECTION_CEILING_EXCEEDED") return {
				status: "unsafe-content",
				reasonCode: "INSPECTION_CEILING_EXCEEDED",
				message: validation.error,
				details: validation.details
			};
			return {
				status: "malformed",
				reasonCode: validation.reasonCode ?? "MALFORMED_METADATA",
				message: validation.error,
				details: validation.details
			};
		}
		let maxLayersInAFrame = 0;
		let totalLayerCount = 0;
		for (let i = 0; i < validation.value.frames.length; i++) {
			const lCount = validation.value.frames[i].layers.length;
			totalLayerCount += lCount;
			if (lCount > maxLayersInAFrame) maxLayersInAFrame = lCount;
		}
		const pxPerFrame = safeMultiply(validation.value.width, validation.value.height) ?? Number.MAX_SAFE_INTEGER;
		const materializableRasterPixels = safeMultiply(totalLayerCount, pxPerFrame) ?? Number.MAX_SAFE_INTEGER;
		const baseRawRgbaBytes = safeMultiply(materializableRasterPixels, 4) ?? Number.MAX_SAFE_INTEGER;
		const metadata = {
			name: validation.value.name,
			width: validation.value.width,
			height: validation.value.height,
			frameCount: validation.value.frames.length,
			totalLayerCount,
			maxLayersInAFrame,
			materializableRasterPixels,
			baseRawRgbaBytes,
			fps: validation.value.fps,
			version: validation.value.version,
			createdAt: validation.value.createdAt,
			updatedAt: validation.value.updatedAt
		};
		if (validation.violations.length === 0) return {
			status: "valid",
			project: validation.value,
			metadata
		};
		return {
			status: "recoverable-envelope-exceeded",
			project: validation.value,
			metadata,
			violations: validation.violations,
			reasonCodes: validation.reasonCodes
		};
	}

		* Maximum allowed total pixel area (67,108,864 pixels = 8192x8192).
		* Prevents allocating multi-gigabyte uncompressed RGBA pixel buffers.
		*/
		static MAX_ALLOWED_PIXELS = WORKSPACE_ENVELOPE.MAX_PROBE_PIXELS;
		/** Maximum bytes to scan during JPEG marker segment traversal (16MB). */
		static MAX_PROBE_SEARCH_WINDOW = 16777216;
		/**
		* Probes an image header from a Blob, File, or raw ArrayBuffer/Uint8Array.
		* Supports streaming window paging for Blob/File inputs to handle late SOF segments.
		*/
		static async probe(input) {
			let headerBytes;
			let chunkStart = 0;
			if (input instanceof Blob) {
				const initialSlice = await input.slice(0, 262144).arrayBuffer();
				headerBytes = new Uint8Array(initialSlice);
			} else if (input instanceof ArrayBuffer) headerBytes = new Uint8Array(input);
			else headerBytes = input;
			if (headerBytes.length < 10) throw new Error("[ImageHeaderProbe] File is truncated or too small to contain a valid image header");
			const view = new DataView(headerBytes.buffer, headerBytes.byteOffset, headerBytes.byteLength);
			if (headerBytes.length >= 20 && view.getUint32(0, false) === 1380533830 && view.getUint32(8, false) === 1464156752) {
				const chunk = view.getUint32(12, false);
				const uint24 = (at) => headerBytes[at] | headerBytes[at + 1] << 8 | headerBytes[at + 2] << 16;
				if (chunk === 1448097880 && headerBytes.length >= 30) return this.validate(uint24(24) + 1, uint24(27) + 1, "WebP");
				if (chunk === 1448097868 && headerBytes.length >= 25 && headerBytes[20] === 47) {
					const bits = view.getUint32(21, true);
					return this.validate((bits & 16383) + 1, (bits >>> 14 & 16383) + 1, "WebP");
				}
				if (chunk === 1448097824 && headerBytes.length >= 30 && headerBytes[23] === 157 && headerBytes[24] === 1 && headerBytes[25] === 42) return this.validate(view.getUint16(26, true) & 16383, view.getUint16(28, true) & 16383, "WebP");
				throw new Error("[ImageHeaderProbe] Unsupported or malformed WebP header");
			}
			if (headerBytes[0] === 137 && headerBytes[1] === 80 && headerBytes[2] === 78 && headerBytes[3] === 71 && headerBytes[4] === 13 && headerBytes[5] === 10 && headerBytes[6] === 26 && headerBytes[7] === 10) {
				if (headerBytes.length < 24) throw new Error("[ImageHeaderProbe] Truncated PNG header: IHDR missing");
				if (String.fromCharCode(headerBytes[12], headerBytes[13], headerBytes[14], headerBytes[15]) !== "IHDR") throw new Error("[ImageHeaderProbe] Malformed PNG: First chunk is not IHDR");
				const width = view.getUint32(16, false);
				const height = view.getUint32(20, false);
				return this.validate(width, height, "PNG");
			}
			if (headerBytes[0] === 71 && headerBytes[1] === 73 && headerBytes[2] === 70 && headerBytes[3] === 56 && (headerBytes[4] === 55 || headerBytes[4] === 57) && headerBytes[5] === 97) {
				if (headerBytes.length < 10) throw new Error("[ImageHeaderProbe] Truncated GIF logical screen descriptor");
				const width = view.getUint16(6, true);
				const height = view.getUint16(8, true);
				return this.validate(width, height, "GIF");
			}
			if (headerBytes[0] === 66 && headerBytes[1] === 77) {
				if (headerBytes.length < 26) throw new Error("[ImageHeaderProbe] Truncated BMP header");
				const width = Math.abs(view.getInt32(18, true));
				const height = Math.abs(view.getInt32(22, true));
				return this.validate(width, height, "BMP");
			}
			if (headerBytes[0] === 255 && headerBytes[1] === 216) {
				const ensureBytes = async (offset, length) => {
					if (offset + length <= chunkStart + headerBytes.length) return true;
					if (input instanceof Blob) {
						if (offset >= input.size) return false;
						const nextChunkSize = Math.max(262144, length);
						chunkStart = offset;
						const slice = await input.slice(chunkStart, chunkStart + nextChunkSize).arrayBuffer();
						headerBytes = new Uint8Array(slice);
						return headerBytes.length >= length;
					}
					return false;
				};
				let fileOffset = 2;
				while (fileOffset < ImageHeaderProbe.MAX_PROBE_SEARCH_WINDOW) {
					if (!await ensureBytes(fileOffset, 2)) throw new Error("[ImageHeaderProbe] SOF_BEYOND_PROBE_WINDOW: Start of frame not found within probe window");
					const markerPrefix = headerBytes[fileOffset - chunkStart];
					fileOffset++;
					if (markerPrefix !== 255) continue;
					if (!await ensureBytes(fileOffset, 1)) throw new Error("[ImageHeaderProbe] SOF_BEYOND_PROBE_WINDOW: Start of frame not found within probe window");
					let marker = headerBytes[fileOffset - chunkStart];
					fileOffset++;
					while (marker === 255) {
						if (!await ensureBytes(fileOffset, 1)) throw new Error("[ImageHeaderProbe] SOF_BEYOND_PROBE_WINDOW: Start of frame not found within probe window");
						marker = headerBytes[fileOffset - chunkStart];
						fileOffset++;
					}
					if (marker === 217 || marker === 218) throw new Error("[ImageHeaderProbe] SOF_BEYOND_PROBE_WINDOW: Start of frame not found within probe window");
					if (marker === 192 || marker === 193 || marker === 194) {
						if (!await ensureBytes(fileOffset, 7)) throw new Error("[ImageHeaderProbe] SOF_BEYOND_PROBE_WINDOW: Start of frame not found within probe window");
						const sofView = new DataView(headerBytes.buffer, headerBytes.byteOffset + (fileOffset - chunkStart), 7);
						const height = sofView.getUint16(3, false);
						const width = sofView.getUint16(5, false);
						return this.validate(width, height, "JPEG");
					}
					if (marker >= 208 && marker <= 215 || marker === 1) continue;
					if (!await ensureBytes(fileOffset, 2)) throw new Error("[ImageHeaderProbe] SOF_BEYOND_PROBE_WINDOW: Start of frame not found within probe window");
					const segmentLength = new DataView(headerBytes.buffer, headerBytes.byteOffset + (fileOffset - chunkStart), 2).getUint16(0, false);
					if (segmentLength < 2) throw new Error("[ImageHeaderProbe] SOF_BEYOND_PROBE_WINDOW: Start of frame not found within probe window");
					fileOffset += segmentLength;
				}
				throw new Error("[ImageHeaderProbe] SOF_BEYOND_PROBE_WINDOW: Start of frame not found within probe window");
			}
			throw new Error("[ImageHeaderProbe] Unsupported or unrecognizable image signature");
		}
		static validate(width, height, format) {
			if (!Number.isSafeInteger(width) || !Number.isSafeInteger(height) || width <= 0 || height <= 0) throw new RangeError(`[ImageHeaderProbe] Corrupted ${format} dimensions declared in header: ${width}x${height}`);
			if (width > this.MAX_ALLOWED_DIMENSION || height > this.MAX_ALLOWED_DIMENSION) throw new RangeError(`[ImageHeaderProbe] Image dimensions (${width}x${height}) exceed maximum allowed workstation limit (${this.MAX_ALLOWED_DIMENSION}px)`);
			if (width * height > this.MAX_ALLOWED_PIXELS) throw new RangeError(`[ImageHeaderProbe] Decompression bomb detected (CWE-409): total pixel area (${width * height}) exceeds safety limit`);
			return {
				width,
				height,
				format
			};
		}
	};

	var INVALID_LAYER_REASON_PRIORITY = {
		CORRUPTED_PNG_HEADER: 1,

	/**
	* Captures an authoritative single bounded snapshot of a project, its complete
	* layer blobs keyspace, and its recovery ledger record in a single read-only transaction.
	*/
	async function acquireStoredProjectSnapshot(projectId, options) {
		if (options?.signal?.aborted) throw new RecoveryAbortError("Recovery snapshot acquisition cancelled.");
		if (typeof projectId !== "string" || projectId.trim().length === 0 || projectId.length > RECOVERY_INSPECTION_CEILINGS.MAX_IDENTIFIER_LENGTH) throw new TypeError(`Invalid projectId: must be non-empty string <= ${RECOVERY_INSPECTION_CEILINGS.MAX_IDENTIFIER_LENGTH}`);
		if (!db.isOpen()) await db.open();
		let projectRecord;
		let sourceBlobs = [];
		let ledgerRecord;
		try {
			await db.transaction("r", [
				db.projects,
				db.layerBlobs,
				db.recoveryRecords
			], async () => {
				if (options?.signal?.aborted) throw new RecoveryAbortError("Recovery snapshot acquisition cancelled.");
				projectRecord = await db.projects.get(projectId);
				if (options?.signal?.aborted) throw new RecoveryAbortError("Recovery snapshot acquisition cancelled.");
				sourceBlobs = await db.layerBlobs.where("[projectId+id]").between([projectId, Dexie.minKey], [projectId, Dexie.maxKey], true, true).limit(RECOVERY_INSPECTION_CEILINGS.MAX_INSPECTABLE_STORED_BLOBS + 1).toArray();
				if (options?.signal?.aborted) throw new RecoveryAbortError("Recovery snapshot acquisition cancelled.");
				ledgerRecord = await db.recoveryRecords.get(projectId);
			});
		} catch (err) {
			if (options?.signal?.aborted || err instanceof RecoveryAbortError || err?.name === "AbortError") throw new RecoveryAbortError("Recovery snapshot acquisition cancelled.");
			throw err;
		}
		return Object.freeze({
			projectRecord,
			sourceBlobs: Object.freeze(sourceBlobs),
			ledgerRecord
		});
	}
	/**
	* Assesses an already captured StoredProjectSnapshot purely in memory.
	* Performs zero database operations.
	*/
	async function assessStoredSnapshot(snapshot, options, fallbackProjectId) {
		const projectId = snapshot.projectRecord?.id ?? fallbackProjectId ?? "unknown";
		if (options?.signal?.aborted) return {
			status: "cancelled",
			projectId,
			message: "Recovery scan was cancelled before execution."
		};
		const projectRecord = snapshot.projectRecord;
		if (!projectRecord) return {
			status: "not-found",
			projectId,
			message: `Project "${projectId}" not found in database.`
		};
		if (!projectRecord.dataJson || typeof projectRecord.dataJson !== "string") return {
			status: "malformed",
			projectId,
			reasonCode: "MALFORMED_METADATA",
			reasonCodes: ["MALFORMED_METADATA"],
			message: "Stored project record has missing or empty dataJson payload."
		};
		const classification = classifyProjectMetadata(projectRecord.dataJson);
		if (classification.status === "malformed") return {
			status: "malformed",
			projectId,
			reasonCode: classification.reasonCode,
			reasonCodes: [classification.reasonCode],
			message: classification.message,
			details: classification.details
		};
		if (classification.status === "unsupported-version") return {
			status: "unsupported-version",
			projectId,
			reasonCode: "UNSUPPORTED_FUTURE_VERSION",
			reasonCodes: ["UNSUPPORTED_FUTURE_VERSION"],
			foundVersion: classification.foundVersion,
			supportedVersion: classification.supportedVersion,
			message: classification.message
		};
		if (classification.status === "unsafe-content") return {
			status: "unsafe-content",
			projectId,
			reasonCode: classification.reasonCode,
			reasonCodes: [classification.reasonCode],
			message: classification.message,
			details: classification.details
		};
		const metadataLayers = /* @__PURE__ */ new Map();
		for (let f = 0; f < classification.project.frames.length; f++) {
			const frame = classification.project.frames[f];
			for (let l = 0; l < frame.layers.length; l++) {
				const layer = frame.layers[l];
				let frameMap = metadataLayers.get(layer.id);
				if (!frameMap) {
					frameMap = /* @__PURE__ */ new Map();
					metadataLayers.set(layer.id, frameMap);
				}
				frameMap.set(f, {
					hasEmbeddedRaster: typeof layer.pngBase64 === "string",
					blobCount: 0,
					blobReasons: []
				});
			}
		}
		if (snapshot.sourceBlobs.length > RECOVERY_INSPECTION_CEILINGS.MAX_INSPECTABLE_STORED_BLOBS) return {
			status: "unsafe-content",
			projectId,
			reasonCode: "EXCESSIVE_STORED_BLOBS",
			reasonCodes: ["EXCESSIVE_STORED_BLOBS"],
			message: `Project exceeds recovery blob inspection ceiling (${RECOVERY_INSPECTION_CEILINGS.MAX_INSPECTABLE_STORED_BLOBS} blobs).`
		};
		let lastSeenId = void 0;
		let scannedBlobs = 0;
		let totalOrphans = 0;
		let duplicateOrphans = 0;
		const orphanDetails = [];
		const orphanSeen = /* @__PURE__ */ new Map();
		for (let i = 0; i < snapshot.sourceBlobs.length; i++) {
			if (options?.signal?.aborted) return {
				status: "cancelled",
				projectId,
				message: "Recovery scan was cancelled."
			};
			const record = snapshot.sourceBlobs[i];
			if (typeof record.id !== "number" || !Number.isSafeInteger(record.id) || record.id <= 0 || lastSeenId !== void 0 && record.id <= lastSeenId) return {
				status: "unsafe-content",
				projectId,
				reasonCode: "CORRUPTED_STORED_BLOB_RECORD",
				reasonCodes: ["CORRUPTED_STORED_BLOB_RECORD"],
				message: "Corrupted stored blob record: primary key is not a positive safe integer or non-monotonic."
			};
			lastSeenId = record.id;
			scannedBlobs++;
			if (!(typeof record.projectId === "string" && record.projectId === projectId && typeof record.layerId === "string" && record.layerId.length > 0 && record.layerId.length <= RECOVERY_INSPECTION_CEILINGS.MAX_IDENTIFIER_LENGTH && typeof record.frameIndex === "number" && Number.isSafeInteger(record.frameIndex) && record.frameIndex >= 0 && record.blob instanceof Blob && typeof record.blob.size === "number" && Number.isSafeInteger(record.blob.size) && record.blob.size >= 0)) return {
				status: "unsafe-content",
				projectId,
				reasonCode: "CORRUPTED_STORED_BLOB_RECORD",
				reasonCodes: ["CORRUPTED_STORED_BLOB_RECORD"],
				message: "Corrupted stored blob record: invalid field types, bounds, or unsafe integers."
			};
			const layerEntry = metadataLayers.get(record.layerId)?.get(record.frameIndex);
			if (!layerEntry) {
				totalOrphans++;
				let seenFrames = orphanSeen.get(record.layerId);
				if (!seenFrames) {
					seenFrames = /* @__PURE__ */ new Set();
					orphanSeen.set(record.layerId, seenFrames);
				}
				if (seenFrames.has(record.frameIndex)) duplicateOrphans++;
				else {
					seenFrames.add(record.frameIndex);
					if (orphanDetails.length < RECOVERY_INSPECTION_CEILINGS.MAX_RECORDED_ORPHAN_DETAILS) orphanDetails.push({
						recordId: record.id,
						ordinal: totalOrphans,
						frameIndex: record.frameIndex,
						sizeBytes: record.blob.size
					});
				}
				continue;
			}
			layerEntry.blobCount++;
			if (layerEntry.blobCount > 1) {
				if (!layerEntry.blobReasons.includes("DUPLICATE_BLOB_MAPPING")) layerEntry.blobReasons.push("DUPLICATE_BLOB_MAPPING");
			} else {
				if (options?.signal?.aborted) return {
					status: "cancelled",
					projectId,
					message: "Recovery scan was cancelled."
				};
				try {
					const slice = record.blob.slice(0, 24);
					const probed = await ImageHeaderProbe.probe(slice);
					if (options?.signal?.aborted) return {
						status: "cancelled",
						projectId,
						message: "Recovery scan was cancelled."
					};
					if (probed.format !== "PNG") layerEntry.blobReasons.push("CORRUPTED_PNG_HEADER");
					else if (probed.width !== classification.project.width || probed.height !== classification.project.height) layerEntry.blobReasons.push("BLOB_DIMENSION_MISMATCH");
				} catch {
					layerEntry.blobReasons.push("CORRUPTED_PNG_HEADER");
				}
			}
			const isBatchBoundary = (i + 1) % RECOVERY_INSPECTION_CEILINGS.BLOB_SCAN_BATCH_SIZE === 0;
			const isLastItem = i === snapshot.sourceBlobs.length - 1;
			if (isBatchBoundary || isLastItem) {
				if (options?.signal?.aborted) return {
					status: "cancelled",
					projectId,
					message: "Recovery scan was cancelled."
				};
				if (options?.onProgress) try {
					options.onProgress({
						scannedBlobs,
						currentBatchSize: isBatchBoundary ? RECOVERY_INSPECTION_CEILINGS.BLOB_SCAN_BATCH_SIZE : snapshot.sourceBlobs.length % RECOVERY_INSPECTION_CEILINGS.BLOB_SCAN_BATCH_SIZE
					});
				} catch {}
				await new Promise((resolve) => setTimeout(resolve, 0));
			}
		}
		let affectedLayerCount = 0;
		let totalReasonCount = 0;
		const invalidLayerDetails = [];
		const collectedReasonCodes = /* @__PURE__ */ new Set();
		if (classification.status === "recoverable-envelope-exceeded") for (const code of classification.reasonCodes) collectedReasonCodes.add(code);
		for (const [layerId, frameMap] of metadataLayers.entries()) for (const [frameIndex, entry] of frameMap.entries()) {
			const reasonsForLayer = [];
			if (entry.hasEmbeddedRaster) reasonsForLayer.push("EMBEDDED_RASTER_UNSUPPORTED");
			for (const r of entry.blobReasons) if (!reasonsForLayer.includes(r)) reasonsForLayer.push(r);
			if (reasonsForLayer.length > 0) {
				reasonsForLayer.sort((a, b) => INVALID_LAYER_REASON_PRIORITY[a] - INVALID_LAYER_REASON_PRIORITY[b]);
				affectedLayerCount++;
				totalReasonCount += reasonsForLayer.length;
				for (const r of reasonsForLayer) collectedReasonCodes.add(r);
				if (invalidLayerDetails.length < RECOVERY_INSPECTION_CEILINGS.MAX_RECORDED_INVALID_LAYERS) invalidLayerDetails.push({
					layerId,
					frameIndex,
					reason: reasonsForLayer[0],
					reasons: reasonsForLayer
				});
			}
		}
		const invalidLayerSummary = {
			affectedLayerCount,
			totalReasonCount,
			detailsTruncated: affectedLayerCount > invalidLayerDetails.length,
			details: invalidLayerDetails
		};
		const orphanSummary = {
			totalCount: totalOrphans,
			duplicateCount: duplicateOrphans,
			detailsTruncated: totalOrphans > orphanDetails.length,
			details: orphanDetails
		};
		if (affectedLayerCount > 0) return {
			status: "recoverable-blob-invalid",
			projectId,
			project: classification.project,
			metadata: classification.metadata,
			invalidLayerSummary,
			violations: classification.status === "recoverable-envelope-exceeded" ? classification.violations : void 0,
			reasonCodes: sortReasonCodes(Array.from(collectedReasonCodes)),
			orphanSummary,
			scannedBlobsCount: scannedBlobs
		};
		if (classification.status === "recoverable-envelope-exceeded") return {
			status: "recoverable-envelope-exceeded",
			projectId,
			project: classification.project,
			metadata: classification.metadata,
			violations: classification.violations,
			reasonCodes: sortReasonCodes(classification.reasonCodes),
			orphanSummary,
			scannedBlobsCount: scannedBlobs
		};
		return {
			status: "valid",
			projectId,
			project: classification.project,
			metadata: classification.metadata,
			orphanSummary,
			scannedBlobsCount: scannedBlobs
		};
	}

	/**
	* Deeply and recursively freezes an object graph to guarantee public runtime immutability.
	* Exempts Blob instances since browser Blobs are natively immutable.
	*/
	function deepFreeze(obj, seen = /* @__PURE__ */ new WeakSet()) {
		if (obj === null || typeof obj !== "object" || Object.isFrozen(obj)) return obj;
		if (typeof Blob !== "undefined" && obj instanceof Blob) return obj;
		if (seen.has(obj)) return obj;
		seen.add(obj);
		for (const key of Object.getOwnPropertyNames(obj)) {
			const value = obj[key];
			if (value !== null && typeof value === "object") deepFreeze(value, seen);
		}
		return Object.freeze(obj);
	}
	/**
	* Authoritative stored-project admission boundary.
	*
	* Accepts a project ID and optional signal only.
	* Internally acquires a single bounded snapshot from IndexedDB, runs the full
	* Phase 1B.2 metadata classification and bounded keyspace Blob assessment,
	* then performs strict validateProjectFile() validation before admitting.
	*
	* Rejects any non-valid assessment before hydration.
	* Returns only frozen, immutable data needed for safe hydration.
	*/
	async function inspectAndAdmitStoredProject(projectId, options) {
		if (options?.signal?.aborted) return { status: "cancelled" };
		if (typeof projectId !== "string" || projectId.trim().length === 0 || projectId.length > RECOVERY_INSPECTION_CEILINGS.MAX_IDENTIFIER_LENGTH) return {
			status: "blocked",
			projectId: String(projectId),
			reasonCode: "INVALID_PROJECT_ID",
			message: `Invalid projectId: must be a non-empty string <= ${RECOVERY_INSPECTION_CEILINGS.MAX_IDENTIFIER_LENGTH}`
		};
		let snapshot;
		try {
			snapshot = await acquireStoredProjectSnapshot(projectId, { signal: options?.signal });
		} catch (err) {
			if (options?.signal?.aborted || err instanceof RecoveryAbortError) return { status: "cancelled" };
			return {
				status: "transient-failure",
				projectId,
				message: err instanceof Error ? err.message : "Database snapshot acquisition failed."
			};
		}
		if (options?.signal?.aborted) return { status: "cancelled" };
		if (!snapshot.projectRecord) return {
			status: "not-found",
			projectId
		};
		let assessment;
		try {
			assessment = await assessStoredSnapshot(snapshot, options);
		} catch (err) {
			if (options?.signal?.aborted || err instanceof RecoveryAbortError) return { status: "cancelled" };
			return {
				status: "transient-failure",
				projectId,
				message: err instanceof Error ? err.message : "Stored snapshot assessment failed."
			};
		}
		if (options?.signal?.aborted || assessment.status === "cancelled") return { status: "cancelled" };
		switch (assessment.status) {
			case "not-found": return {
				status: "not-found",
				projectId
			};
			case "transient-load-failure": return {
				status: "transient-failure",
				projectId,
				message: assessment.message
			};
			case "unsupported-version": return {
				status: "unsupported-version",
				projectId,
				foundVersion: assessment.foundVersion,
				supportedVersion: assessment.supportedVersion,
				message: assessment.message
			};
			case "malformed": return {
				status: "blocked",
				projectId,
				reasonCode: assessment.reasonCode,
				message: assessment.message
			};
			case "unsafe-content": return {
				status: "blocked",
				projectId,
				reasonCode: assessment.reasonCode,
				message: assessment.message
			};
			case "recoverable-envelope-exceeded":
			case "recoverable-blob-invalid": return {
				status: "recovery-required",
				projectId,
				assessment
			};
			case "valid": {
				const strictResult = validateProjectFile(assessment.project);
				if (!strictResult.ok) return {
					status: "blocked",
					projectId,
					reasonCode: "STRICT_VALIDATION_FAILED",
					message: strictResult.error
				};
				const frozenBlobs = Object.freeze(snapshot.sourceBlobs.map((b) => Object.freeze({
					id: b.id,
					projectId: b.projectId,
					layerId: b.layerId,
					frameIndex: b.frameIndex,
					blob: b.blob,
					updatedAt: b.updatedAt
				})));
				return {
					status: "admitted",
					project: deepFreeze(structuredClone(strictResult.value)),
					projectId,
					sourceBlobs: frozenBlobs,
					storageVersion: getStoredProjectVersion(snapshot.projectRecord)
				};
			}
		}
	}
