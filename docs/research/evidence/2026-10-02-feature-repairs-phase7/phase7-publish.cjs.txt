const fs = require('node:fs');
const path = require('node:path');
const crypto = require('node:crypto');
const source = 'artifacts/feature-repairs/phase7';
const destination = 'docs/research/evidence/2026-10-02-feature-repairs-phase7';
const sha = bytes => crypto.createHash('sha256').update(bytes).digest('hex');
const mode = process.argv[2];
if (mode === 'stage') {
  if (fs.existsSync(destination)) throw new Error('Refusing to overwrite an evidence directory');
  fs.mkdirSync(destination);
  for (const file of fs.readdirSync(source, { withFileTypes: true })) {
    if (!file.isFile()) continue;
    if (/\.(?:log|json|txt)$/.test(file.name)) fs.copyFileSync(path.join(source, file.name), path.join(destination, file.name));
  }
  for (const name of ['phase7-run.cjs', 'phase7-release-check.cjs', 'phase7-rollback-bundle.cjs', 'phase7-rollback-check.cjs', 'phase7-live-smoke.cjs', 'phase7-npm.cjs', 'phase7-prepare-editor.cjs', 'phase7-clean-editor.cjs', 'phase7-build-inputs.cjs', 'phase7-integrity.cjs', 'phase7-candidate-http.cjs', 'phase7-trace-inspect.py', 'phase6-zip-oracle.py']) fs.copyFileSync(`artifacts/feature-repairs/${name}`, `${destination}/${name}.txt`);
  for (const name of ['historical-reader.js', 'current-reader.js', 'diagnostic-reader-entry.mts', 'playwright.editor.config.cjs']) fs.copyFileSync(`${source}/${name}`, `${destination}/${name}.txt`);
  fs.mkdirSync(`${destination}/theme-navigation-failure`);
  const failure = `${source}/editor-final/themeAccessibility-Four-Th-6c32b-rds-under-Cozy-Classic-Dark-chromium`;
  fs.copyFileSync(`${failure}/trace.zip`, `${destination}/theme-navigation-failure/trace.zip`);
  fs.copyFileSync(`${failure}/error-context.md`, `${destination}/theme-navigation-failure/context.txt`);
  console.log('Staged a new unsealed phase 7 evidence directory.');
} else if (mode === 'seal') {
  if (fs.existsSync(`${destination}/manifest.json`)) throw new Error('Evidence is already sealed');
  const files = [];
  function walk(directory) {
    for (const entry of fs.readdirSync(directory, { withFileTypes: true })) {
      const full = path.join(directory, entry.name);
      if (entry.isDirectory()) walk(full);
      else files.push({ path: path.relative(destination, full).replaceAll('\\', '/'), sha256: sha(fs.readFileSync(full)) });
    }
  }
  walk(destination); files.sort((a,b) => a.path.localeCompare(b.path));
  fs.writeFileSync(`${destination}/manifest.json`, JSON.stringify(files, null, 2) + '\n');
  for (const file of files) if (sha(fs.readFileSync(path.join(destination, file.path))) !== file.sha256) throw new Error(`Invalid sealed hash: ${file.path}`);
  console.log(`Sealed and verified ${files.length} phase 7 evidence files.`);
} else throw new Error('Expected stage or seal');
