const fs = require('node:fs');
const dns = require('node:dns/promises');
const { chromium, expect } = require('@playwright/test');
const folder = 'artifacts/feature-repairs/phase7';
(async () => {
  const result = { recordedAt: new Date().toISOString(), kind: 'published-site-read-only-smoke-not-candidate-release-qualification', origin: 'https://pixibun.vercel.app', consoleErrors: [], pageErrors: [] };
  try { result.customDomainLookup = await dns.lookup('pixibun.art', { all: true }); }
  catch (error) { result.customDomainLookup = { code: error.code, message: error.message }; }
  const browser = await chromium.launch();
  try {
    result.browserVersion = browser.version();
    const context = await browser.newContext(); const page = await context.newPage();
    page.on('console', message => { if (message.type() === 'error') result.consoleErrors.push(message.text()); });
    page.on('pageerror', error => result.pageErrors.push(String(error)));
    const response = await page.goto(result.origin + '/studio', { timeout: 45_000 });
    result.studioStatus = response.status();
    await expect(page.locator('[data-restoration-phase="active"]')).toBeVisible({ timeout: 20_000 });
    await expect(page.getByTestId('main-composite-canvas')).toBeVisible();
    await expect.poll(() => page.evaluate(() => Boolean(navigator.serviceWorker.controller)), { timeout: 30_000 }).toBe(true);
    result.online = await page.evaluate(async () => ({ controller: navigator.serviceWorker.controller.scriptURL, caches: await caches.keys(), dimensions: { width: document.querySelector('[data-testid="main-composite-canvas"]').width, height: document.querySelector('[data-testid="main-composite-canvas"]').height } }));
    await expect.poll(() => page.evaluate(async () => {
      const urls = performance.getEntriesByType('resource').map(e => e.name).filter(u => new URL(u).pathname.startsWith('/_next/static/'));
      return (await Promise.all(urls.map(async u => await caches.match(u) ? null : u))).filter(Boolean);
    }), { timeout: 20_000 }).toEqual([]);
    await context.setOffline(true); await page.reload();
    await expect(page.locator('[data-restoration-phase="active"]')).toBeVisible();
    await expect(page.getByTestId('main-composite-canvas')).toBeVisible();
    result.warmOfflineReload = true;
    // An offline update fetch can legitimately fail; classify the observable console output separately.
    result.verdict = result.pageErrors.length ? 'failed-page-error' : 'passed-studio-worker-and-warm-offline-smoke';
    await context.close();
  } catch (error) { result.verdict = 'failed-or-inconclusive'; result.error = String(error); process.exitCode = 1; }
  finally { await browser.close(); fs.writeFileSync(`${folder}/live-smoke.json`, JSON.stringify(result, null, 2) + '\n'); console.log(JSON.stringify(result, null, 2)); }
})().catch(error => { console.error(error); process.exitCode = 1; });
