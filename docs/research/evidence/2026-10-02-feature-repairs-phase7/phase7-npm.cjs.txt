const fs = require('node:fs');
const cp = require('node:child_process');
const path = require('node:path');
const folder = 'artifacts/feature-repairs/phase7';
const mode = process.argv[2];
const args = mode === 'patch' ? ['install', 'next@16.3.6', '--save-exact', '--ignore-scripts', '--no-audit', '--no-fund'] : ['audit', '--omit=dev', '--json'];
const npm = path.join(path.dirname(process.execPath), 'node_modules/npm/bin/npm-cli.js');
const log = fs.openSync(`${folder}/${mode === 'patch' ? 'dependency-patch.log' : 'dependency-audit-final.json'}`, 'wx');
const stderr = fs.openSync(`${folder}/npm-${mode}-stderr.log`, 'wx');
const env = { ...process.env, PATH: `${path.dirname(process.execPath)}${path.delimiter}${process.env.PATH}` };
const start = Date.now();
const child = cp.spawn(process.execPath, [npm, ...args], { env, stdio: ['ignore', log, stderr] });
child.on('error', error => { console.error(error); process.exitCode = 1; });
child.on('exit', (code, signal) => {
  fs.closeSync(log); fs.closeSync(stderr);
  const result = { recordedAt: new Date().toISOString(), command: ['node', npm, ...args], exitCode: code, signal, elapsedSeconds: (Date.now() - start) / 1000, npmVersion: cp.execFileSync(process.execPath, [npm, '--version'], { encoding: 'utf8' }).trim() };
  fs.writeFileSync(`${folder}/npm-${mode}-result.json`, JSON.stringify(result, null, 2) + '\n');
  console.log(JSON.stringify(result)); process.exitCode = code ?? 1;
});
