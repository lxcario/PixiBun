const fs = require('node:fs');
const { chromium } = require('playwright');
(async () => {
  const browser = await chromium.launch();
  try {
    const page = await browser.newPage();
    const base64 = fs.readFileSync('artifacts/feature-repairs/ci-followup/8cd1ce844a8b59b8b47e4db26d6fb3880a005dcc.mp4').toString('base64');
    const result = await page.evaluate(async base64 => {
      const bytes = Uint8Array.from(atob(base64), c => c.charCodeAt(0));
      const url = URL.createObjectURL(new Blob([bytes], { type: 'video/mp4' }));
      const video = document.createElement('video'); video.muted = true;
      await new Promise((resolve, reject) => { video.onloadeddata = resolve; video.onerror = () => reject(new Error(video.error?.message)); video.src = url; });
      const canvas = document.createElement('canvas'); canvas.width = video.videoWidth; canvas.height = video.videoHeight;
      const ctx = canvas.getContext('2d');
      const pixel = () => { ctx.drawImage(video, 0, 0); return [...ctx.getImageData(canvas.width / 2, canvas.height / 2, 1, 1).data]; };
      const initial = pixel();
      const presented = new Promise(resolve => video.requestVideoFrameCallback((_time, info) => resolve(info.mediaTime)));
      await new Promise(resolve => { video.onseeked = resolve; video.currentTime = 0.12; });
      const afterSeek = pixel();
      const mediaTime = await Promise.race([presented, new Promise(resolve => setTimeout(() => resolve('no callback'), 3000))]);
      const afterPresentation = pixel();
      URL.revokeObjectURL(url);
      return { width: canvas.width, height: canvas.height, duration: video.duration, initial, afterSeek, afterPresentation, mediaTime };
    }, base64);
    console.log(JSON.stringify(result));
  } finally { await browser.close(); }
})().catch(error => { console.error(error); process.exitCode = 1; });
