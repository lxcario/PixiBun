// @vitest-environment happy-dom
// Isolated desired-behavior diagnostics, outside the normal regression inventory.
import { beforeEach, afterEach, it, expect, vi } from 'vitest';
import { makeCanvas } from '../../tests/helpers/mockCanvas';
import { useStudioStore } from '@/stores/studioStore';
import { clearAllBuffers, getOrCreateBuffer, snapshotBuffers } from '@/lib/engine/bufferRegistry';
import { createHistoryEngine, setGlobalHistoryEngine } from '@/lib/engine/historyEngine';
import { restoreStructuralWorkspace } from '@/lib/engine/structuralHistory';
import { applyResearchPreview, researchWorkspaceKey, type ResearchPreview } from '@/lib/research/workbench';
let unsubscribe: (() => void) | undefined;
beforeEach(() => {
  const contexts = new WeakMap<HTMLCanvasElement, CanvasRenderingContext2D>();
  vi.spyOn(HTMLCanvasElement.prototype, 'getContext').mockImplementation(function(this: HTMLCanvasElement) {
    let ctx = contexts.get(this); if (!ctx) { ctx = makeCanvas(this.width || 32, this.height || 32).getContext('2d')!; contexts.set(this, ctx); } return ctx;
  });
  clearAllBuffers(); setGlobalHistoryEngine(null); useStudioStore.getState().resetToNewProject('publication-probe');
  const s = useStudioStore.getState(), b = getOrCreateBuffer(s.activeLayerId, 32, 32);
  b.ctx.fillStyle = '#ff0000'; b.ctx.fillRect(0, 0, 2, 2);
});
afterEach(() => { unsubscribe?.(); unsubscribe = undefined; setGlobalHistoryEngine(null); clearAllBuffers(); vi.restoreAllMocks(); });
function rejectPublication() {
  const initialFrames = useStudioStore.getState().frames;
  let fired = false;
  unsubscribe = useStudioStore.subscribe(state => { if (!fired && state.frames !== initialFrames) { fired = true; throw new Error('Injected subscriber failure'); } });
}
it('restores both prior metadata and raster ownership when a structural publication fails', () => {
  const before = useStudioStore.getState(), buffers = snapshotBuffers();
  const next = structuredClone(before.frames);
  next[0].layers[0].id = 'replacement-layer'; next[0].layers[0].name = 'Replacement';
  const imageData = getOrCreateBuffer(before.activeLayerId, 32, 32).ctx.getImageData(0, 0, 32, 32);
  rejectPublication();
  expect(() => restoreStructuralWorkspace({ canvasWidth: 32, canvasHeight: 32, frames: next, activeFrameIndex: 0,
    activeLayerId: 'replacement-layer', layerBuffers: [{ layerId: 'replacement-layer', imageData }] })).toThrow('Injected subscriber failure');
  console.log(JSON.stringify({ probe: 'structural', frameLayerIds: useStudioStore.getState().frames[0].layers.map(l => l.id),
    bufferIds: [...snapshotBuffers().keys()], previousLayerId: before.activeLayerId }));
  expect([...snapshotBuffers().keys()]).toEqual([...buffers.keys()]);
  expect(useStudioStore.getState().frames).toEqual(before.frames);
});
it('rolls back generated metadata, buffers and history when publication fails', () => {
  const before = useStudioStore.getState(), buffers = snapshotBuffers(), engine = createHistoryEngine();
  setGlobalHistoryEngine(engine);
  const data = getOrCreateBuffer(before.activeLayerId, 32, 32).ctx.getImageData(0, 0, 32, 32).data;
  const dispose = vi.fn();
  const preview: ResearchPreview = { key: researchWorkspaceKey(), name: 'Probe', width: 32, height: 32, dispose,
    frames: [{ index: 0, sourceId: before.activeLayerId, replacedId: before.activeLayerId, before: data,
      recipe: { version: 1, name: 'Probe', width: 32, height: 32,
        settings: { kind: 'pixelize', block: 4, candidate: 'average', colors: ['#000000', '#ffffff'], protected: [] },
        source: { frameId: before.frames[0].id, layerId: before.activeLayerId, sha256: '0'.repeat(64) } },
      result: { data: data.slice(), notes: [] } }] };
  rejectPublication();
  expect(() => applyResearchPreview(preview)).toThrow('Injected subscriber failure');
  console.log(JSON.stringify({ probe: 'generated', layersBefore: before.frames[0].layers.length,
    layersAfter: useStudioStore.getState().frames[0].layers.length, buffersBefore: buffers.size,
    buffersAfter: snapshotBuffers().size, historyDepth: engine.depth, disposed: dispose.mock.calls.length }));
  expect(useStudioStore.getState().frames).toEqual(before.frames);
  expect([...snapshotBuffers().keys()]).toEqual([...buffers.keys()]);
  expect(engine.depth).toBe(0);
  expect(dispose).toHaveBeenCalledOnce();
});
