import 'fake-indexeddb/auto';
import Dexie from 'dexie';
import { it, expect } from 'vitest';
import { db } from '@/lib/storage/db';
import { prepareUntouchedRecoveryArchive } from '@/lib/recovery/recoveryBackup';
import { RECOVERY_INSPECTION_CEILINGS } from '@/lib/recovery/recoveryLimits';

it('diagnostic only: measures recovery overflow fixture phases', async () => {
  await db.projects.clear();
  await db.layerBlobs.clear();
  await db.recoveryRecords.clear();
  const id = 'phase6-timing';
  await db.projects.put({ id, name: 'Timing', width: 32, height: 32, fps: 12, createdAt: 1, updatedAt: 1, dataJson: '{}' });
  const start = performance.now();
  const rows = Array.from({ length: RECOVERY_INSPECTION_CEILINGS.MAX_INSPECTABLE_STORED_BLOBS + 1 }, (_, i) => ({
    id: i + 1, projectId: id, layerId: `layer-${i + 1}`, frameIndex: 0,
    blob: new Blob(['png'], { type: 'image/png' }), updatedAt: 1000,
  }));
  const built = performance.now();
  await db.layerBlobs.bulkPut(rows);
  const seeded = performance.now();
  const collection = () => db.layerBlobs.where('[projectId+id]')
    .between([id, Dexie.minKey], [id, Dexie.maxKey], true, true)
    .limit(RECOVERY_INSPECTION_CEILINGS.MAX_INSPECTABLE_STORED_BLOBS + 1);
  expect(await collection().count()).toBe(rows.length);
  const counted = performance.now();
  expect(await collection().toArray()).toHaveLength(rows.length);
  const enumerated = performance.now();
  await expect(prepareUntouchedRecoveryArchive(id)).rejects.toMatchObject({ code: 'EXCESSIVE_STORED_BLOBS' });
  const prepared = performance.now();
  process.stdout.write(JSON.stringify({ kind: 'diagnostic-not-latency-qualification', records: rows.length,
    fixtureBuildMs: built - start, seedMs: seeded - built, boundedCountMs: counted - seeded,
    payloadEnumerationMs: enumerated - counted, archivePrepareMs: prepared - enumerated }) + '\n');
}, 30_000);
